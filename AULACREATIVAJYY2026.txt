  <!-- LÓGICA DEL SISTEMA -->
  <script type="module">
    import { initializeApp } from "https://www.gstatic.com/firebasejs/10.8.1/firebase-app.js";
    import { getAuth, signInAnonymously, signInWithCustomToken, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/10.8.1/firebase-auth.js";
    import { getFirestore, collection, addDoc, onSnapshot, deleteDoc, doc, query, orderBy } from "https://www.gstatic.com/firebasejs/10.8.1/firebase-firestore.js";

    feather.replace();

    let app, auth, db;
    let rutaMateriales = "materiales_v2"; // Nueva colección para organizar por categoría
    let rutaForo = "foro_general";
    let usuarioActual = null;
    window.esAdmin = false; // Control de seguridad en pantalla

    // --- CONFIGURACIÓN DE NUBE AUTOMÁTICA ---
    if (typeof __firebase_config !== 'undefined') {
        const firebaseConfig = JSON.parse(__firebase_config);
        app = initializeApp(firebaseConfig);
        const appId = typeof __app_id !== 'undefined' ? __app_id : 'default-app-id';
        rutaMateriales = `artifacts/${appId}/public/data/materiales_v2`;
        rutaForo = `artifacts/${appId}/public/data/foro_general`;
    } else {
        // Fallback genérico para cuando lo descargues
        app = initializeApp({
          apiKey: "fake-key-for-local-test", 
          projectId: "aulacreativajy2026"
        });
    }

    auth = getAuth(app);
    db = getFirestore(app);

    let todosLosMateriales = [];
    let mensajesForo = [];

    // AUTENTICACIÓN
    const iniciarSesion = async () => {
      try {
        if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
            await signInWithCustomToken(auth, __initial_auth_token);
        } else {
            await signInAnonymously(auth);
        }
      } catch (err) {
        document.getElementById('estado-conexion').innerHTML = `<i data-feather="alert-triangle" class="w-3 h-3 text-red-500"></i> <span class="text-red-400">Sin conexión</span>`;
        feather.replace();
      }
    };
    iniciarSesion();

    onAuthStateChanged(auth, (user) => {
        if (user) {
            usuarioActual = user;
            document.getElementById('estado-conexion').innerHTML = `<i data-feather="wifi" class="w-3 h-3 text-emerald-400"></i> <span class="text-emerald-400">Conectado</span>`;
            feather.replace();
            iniciarLecturaDeDatos();
        }
    });

    function iniciarLecturaDeDatos() {
      // Leer Materiales
      onSnapshot(collection(db, rutaMateriales), (snapshot) => {
        todosLosMateriales = snapshot.docs.map(d => ({id: d.id, ...d.data()}));
        todosLosMateriales.sort((a,b) => b.fecha.localeCompare(a.fecha));
        renderizarMateriales();
      });

      // Leer Foro de Comunidad
      onSnapshot(collection(db, rutaForo), (snapshot) => {
        mensajesForo = snapshot.docs.map(d => ({id: d.id, ...d.data()}));
        mensajesForo.sort((a,b) => a.fecha.localeCompare(b.fecha)); // Viejos arriba, nuevos abajo
        renderizarForo();
      });
    }

    // RENDERIZAR TABS DE MATERIAL
    function renderizarMateriales() {
      const listas = {
        planeacion: document.getElementById('lista-planeacion'),
        ova: document.getElementById('lista-ovas'),
        libro: document.getElementById('lista-libro')
      };

      // Limpiar contenedores
      Object.values(listas).forEach(l => l.innerHTML = '');

      todosLosMateriales.forEach((mat) => {
        const contenedor = listas[mat.categoria];
        if(!contenedor) return;

        const iconMap = { planeacion: 'file-text', ova: 'video', libro: 'book' };
        
        const div = document.createElement('div');
        div.className = "bg-[#1e1541]/80 border border-indigo-500/20 p-5 rounded-xl transition-all shadow-md flex items-center justify-between";
        div.innerHTML = `
          <div class="flex items-center gap-4 overflow-hidden">
            <div class="bg-indigo-900 p-3 rounded-lg text-indigo-300 shrink-0">
               <i data-feather="${iconMap[mat.categoria] || 'file'}" class="w-6 h-6"></i>
            </div>
            <div class="truncate pr-4">
              <h4 class="font-bold text-white text-lg truncate">${mat.titulo}</h4>
              <a href="${mat.url}" target="_blank" class="text-sm text-purple-400 hover:text-purple-300 flex items-center gap-1 mt-1 transition-colors">
                <i data-feather="external-link" class="w-3 h-3"></i> Abrir documento
              </a>
            </div>
          </div>
          
          <button onclick="eliminarMaterial('${mat.id}')" class="admin-only ${window.esAdmin ? '' : 'hidden'} p-2 text-red-400 hover:bg-red-500/20 rounded-lg shrink-0" title="Borrar (Solo Profe)">
            <i data-feather="trash-2" class="w-5 h-5"></i>
          </button>
        `;
        contenedor.appendChild(div);
      });

      // Rellenar vacíos
      Object.keys(listas).forEach(cat => {
        if(listas[cat].children.length === 0) {
          listas[cat].innerHTML = `<div class="col-span-full py-8 text-center text-indigo-300/50">Aún no hay contenido en esta sección.</div>`;
        }
      });

      feather.replace();
    }

    // RENDERIZAR FORO
    function renderizarForo() {
      const caja = document.getElementById('lista-comunidad');
      caja.innerHTML = '';
      if(mensajesForo.length === 0) {
        caja.innerHTML = `<div class="text-center text-indigo-300/50 mt-10">Sé el primero en saludar 👋</div>`;
        return;
      }

      mensajesForo.forEach(msg => {
        const div = document.createElement('div');
        div.className = "bg-[#130d26] p-3 rounded-lg border border-indigo-500/20 w-max max-w-[85%]";
        div.innerHTML = `
          <div class="text-xs font-bold text-fuchsia-400 mb-1 flex items-center gap-2">
            ${msg.autor}
            ${window.esAdmin ? `<button onclick="eliminarForo('${msg.id}')" class="text-red-500 hover:text-red-300"><i data-feather="x" class="w-3 h-3"></i></button>` : ''}
          </div>
          <div class="text-sm text-indigo-100">${msg.texto}</div>
        `;
        caja.appendChild(div);
      });
      // Scroll abajo
      caja.scrollTop = caja.scrollHeight;
      feather.replace();
    }

    // INTERFAZ Y NAVEGACIÓN
    window.cambiarPestana = (pestana) => {
      ['inicio', 'planeacion', 'ovas', 'libro', 'comunidad'].forEach(p => {
        document.getElementById(`view-${p}`).classList.add('hidden');
        document.getElementById(`view-${p}`).classList.remove('flex'); // Para comunidad
      });

      document.querySelectorAll('.nav-btn').forEach(btn => {
        btn.className = "nav-btn w-full flex items-center justify-center md:justify-start gap-3 p-3 rounded-lg transition-all text-indigo-300 hover:bg-white/5 hover:text-white";
      });

      const btnActivo = document.querySelector(`[data-tab="${pestana}"]`);
      if(btnActivo) btnActivo.className = "nav-btn w-full flex items-center justify-center md:justify-start gap-3 p-3 rounded-lg transition-all bg-purple-600/20 text-purple-300 border border-purple-500/30";

      const view = document.getElementById(`view-${pestana}`);
      view.classList.remove('hidden');
      if(pestana === 'comunidad') view.classList.add('flex');
    };

    window.mostrarMensaje = (texto, error = true) => {
      const msjBox = document.getElementById('mensaje-alerta');
      document.getElementById('mensaje-texto').innerText = texto;
      msjBox.className = `fixed top-5 right-5 px-6 py-3 rounded-lg shadow-xl z-50 flex items-center gap-2 fade-in text-white ${error ? 'bg-red-500' : 'bg-emerald-500'}`;
      setTimeout(() => msjBox.classList.add('hidden'), 3500);
    };

    // MODO PROFESOR (LA MAGIA DE LA SEGURIDAD)
    window.activarModoProfesor = () => {
      if(window.esAdmin) {
        window.esAdmin = false;
        document.querySelectorAll('.admin-only').forEach(el => el.classList.add('hidden'));
        document.getElementById('btn-modo-profe').innerHTML = `<i data-feather="lock" class="w-4 h-4"></i> <span class="hidden md:inline">Soy Profesor</span>`;
        document.getElementById('btn-modo-profe').classList.remove('bg-fuchsia-600', 'text-white');
        mostrarMensaje("Modo profesor desactivado.", false);
      } else {
        const pass = prompt("Introduce la clave de acceso para profesores:");
        if (pass === "profe2026") {
          window.esAdmin = true;
          document.querySelectorAll('.admin-only').forEach(el => el.classList.remove('hidden'));
          document.getElementById('btn-modo-profe').innerHTML = `<i data-feather="unlock" class="w-4 h-4"></i> <span class="hidden md:inline">Modo Profe Activo</span>`;
          document.getElementById('btn-modo-profe').classList.add('bg-fuchsia-600', 'text-white');
          mostrarMensaje("¡Acceso concedido! Ya puedes subir y borrar archivos.", false);
        } else if(pass !== null) {
          mostrarMensaje("Clave incorrecta.", true);
        }
      }
      feather.replace();
    };

    // SUBIDA UNIVERSAL
    window.subirMaterialGlobal = async (e, categoria) => {
      e.preventDefault();
      if(!usuarioActual) return mostrarMensaje("Esperando conexión a la nube...", true);

      const titulo = document.getElementById(`titulo-${categoria}`).value;
      const inputArchivo = document.getElementById(`archivo-${categoria}`);
      const archivo = inputArchivo.files[0];
      
      // Ovas tiene link opcional
      let linkExterno = '';
      if(categoria === 'ova') {
         linkExterno = document.getElementById('link-ova').value;
      }

      if (!titulo || (!linkExterno && !archivo)) {
        return mostrarMensaje("Completa los datos necesarios.", true);
      }

      const btn = document.getElementById(`btn-${categoria}`);
      btn.disabled = true;
      btn.innerText = "Cargando...";

      let enlaceFinal = linkExterno;

      try {
        if (archivo) {
          const data = new FormData();
          data.append("file", archivo);
          data.append("upload_preset", "Archivosjy"); 

          const res = await fetch("https://api.cloudinary.com/v1_1/dz82ld7bc/auto/upload", {
            method: "POST",
            body: data,
          });
          const archivoSubido = await res.json();
          enlaceFinal = archivoSubido.secure_url; 
        }

        await addDoc(collection(db, rutaMateriales), {
          titulo: titulo,
          url: enlaceFinal,
          categoria: categoria,
          fecha: new Date().toISOString()
        });

        e.target.reset(); // Limpia el formulario
        mostrarMensaje("¡Subido correctamente!", false);

      } catch (error) {
        mostrarMensaje("Error al subir archivo.", true);
      } finally {
        btn.disabled = false;
        btn.innerText = "Publicar";
      }
    };

    window.eliminarMaterial = async (id) => {
      if(!window.esAdmin) return;
      if(window.confirm("¿Borrar definitivamente este archivo de la plataforma?")) {
        await deleteDoc(doc(db, rutaMateriales, id));
      }
    };

    // FUNCIONES DEL FORO
    window.enviarMensajeForo = async (e) => {
      e.preventDefault();
      if(!usuarioActual) return;
      
      const nombre = document.getElementById('nombre-foro').value;
      const texto = document.getElementById('texto-foro').value;

      try {
        await addDoc(collection(db, rutaForo), {
          autor: nombre,
          texto: texto,
          fecha: new Date().toISOString()
        });
        document.getElementById('texto-foro').value = '';
      } catch(err) {
        mostrarMensaje("Error al enviar mensaje.", true);
      }
    };

    window.eliminarForo = async (id) => {
      if(window.confirm("¿Borrar este comentario?")) {
        await deleteDoc(doc(db, rutaForo, id));
      }
    };

  </script>
